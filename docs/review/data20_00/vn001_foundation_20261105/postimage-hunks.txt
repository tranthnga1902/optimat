class FoundationItem(BaseModel):
    """Một loại móng: n khối dài × rộng × cao, hoặc các lớp (tiers), hoặc thể tích có sẵn (rough/foundations.py)."""
    name: str                                       # M1, móng băng trục A...
    count: int = 1
    kind: Optional[Literal["don", "bang"]] = None    # móng đơn | móng băng (dài = chiều dài băng đã trừ phần giao)
    gross_length_mm: Optional[float] = None          # chiều dài toàn băng trên chi tiết, chờ đối chiếu giao trước khi tính
    length_mm: Optional[float] = None
    width_mm: Optional[float] = None
    height_mm: Optional[float] = None
    tiers: list[FootingTier] = Field(default_factory=list)   # có lớp thì thắng dài × rộng × cao
    volume_m3: Optional[float] = None               # nhập thẳng thể tích một khối (thắng kích thước)
    lean_thickness_mm: Optional[float] = None       # bê tông lót dưới móng, cùng mặt bằng móng
    bottom_level_mm: Optional[float] = None         # cao độ mặt dưới khối móng (trên lớp lót) so với ±0.000
    neck_b_mm: Optional[float] = None               # tiết diện cổ móng (phần chôn dưới đất được trừ khi đắp lại)
    neck_h_mm: Optional[float] = None
    positions: list[tuple[float, float]] = Field(default_factory=list)   # tâm từng khối (mm, tọa độ mặt bằng móng)
    plan: Optional[str] = None                      # mặt bằng móng chứa positions (để cắt giằng qua móng)
    source: str = ""                                # truy ngược: file, hình chi tiết, mặt cắt đã đọc

def _footing_rows(foundations: list[FoundationItem]):
    """Một ký hiệu có thể xuất hiện ở nhiều vị trí hoặc trong hai chi tiết mâu thuẫn."""
    groups: dict[str, list[FoundationItem]] = {}
    warnings = []
    for f in foundations:
        key = f.name.strip().upper()
        group = groups.setdefault(key, [])
        if any(f.model_copy(update={"name": prev.name}) == prev for prev in group):
            warnings.append(f"Móng {f.name} nhập hai lần với cùng dữ liệu và nguồn: chỉ giữ một dòng.")
        else:
            group.append(f)

    rows = []
    for key, group in groups.items():
        issue = ""
        if len(group) > 1:
            located = all(f.plan and f.plan == group[0].plan and len(f.positions) == f.count and f.count > 0 for f in group)
            if not located or any(math.dist(p, q) < SAME_SPOT_MM for i, f in enumerate(group)
                                  for other in group[:i] for p in f.positions for q in other.positions):
                issue = f"Móng {key} có nhiều dòng cùng ký hiệu: cần đối chiếu vị trí từng khối trên cùng mặt bằng."
                warnings.append(issue)
        for i, f in enumerate(group, 1):
            rows.append((f, key if len(group) == 1 else f"{key}:{i}", issue))
    return rows, warnings

                                formula=f"−{_fmt(extra)} m³ (đất đắp trong phần hố chung)"))

    return out, warnings

def footing_rects(foundations: list[FoundationItem], plan: str) -> list[tuple[str, tuple[float, float, float, float]]]:
    """Khung mặt bằng (lớp trên cùng, nơi giằng gối vào) của từng khối móng có vị trí trên mặt bằng móng `plan`.
    shortcut: coi cạnh dài theo trục x như pit_overlap_m3; móng xoay 90° cần thêm hướng vào positions."""
    out = []
    rows, _ = _footing_rows(foundations)
    for f, _, issue in rows:
        tiers = tiers_of(f)
        if issue or f.plan != plan or not tiers:
            continue
        t = tiers[-1]
        hl, hw = (t.top_length_mm or t.length_mm) / 2, (t.top_width_mm or t.width_mm) / 2
        out += [(f.name, (x - hl, y - hw, x + hl, y + hw)) for x, y in f.positions]
    return out

def element_of(text: str) -> tuple[str, str, str] | None:
    """Ký hiệu cấu kiện -> (vai trò, tên loại, khóa chữ thường bỏ dấu). Chữ phải khớp trọn mẫu."""
    key = re.sub(r"\s+", "", normalize(text))
    key = re.sub(r"(?<=[a-z])[-–](?=\d)", "", key)
    for pat, role, name in load_element_rules():
        if pat.fullmatch(key):
            return role, name, key
    return None

def footing_detail_sizes(views: list[View], sections: dict, dims: list[Dim]) -> dict[str, dict]:
    """Kích thước móng từ chi tiết: {khóa móng: {length_mm, width_mm, height_mm, lean_thickness_mm, nguon}}."""
    out = {}
    details = [v for v in views if v.ma == "BV05" and v.bbox]
    for v in details:
        k = _title_key(v.title)
        if not k or k[0] != "FOUNDATION":
            continue
        if k[1].startswith("bm"):
            # Hình cắt ngang móng băng không cho chiều dài chạy dọc băng.
            continue
        # khung hình nới xuống tới dưới tiêu đề (đường kích thước tổng hay nằm sát tiêu đề), mỗi bên không quá nửa
        # khoảng cách tới chi tiết móng gần nhất cùng hàng (khung tự tìm có thể lấn sang hình bên cạnh)
        x0, y0, x1, y1 = _grown(v.bbox)
        gaps = [abs(o.x - v.x) for o in details if o is not v and abs(o.y - v.y) < 5000]
        if gaps:
            x0, x1 = max(x0, v.x - min(gaps) / 2), min(x1, v.x + min(gaps) / 2)
        ds = inside(dims, (x0, min(y0, v.y - 1000), x1, y1))
        horiz = [d.value for d in ds if not d.vertical]
        vert = [d.value for d in ds if d.vertical]
        secs = [d for s in sections.get(k, []) for d in s if d.vertical]
        heights = [d.value for d in secs if FOOTING_H_MM[0] <= d.value <= FOOTING_H_MM[1]]
        leans = [d.value for d in secs if LEAN_MM[0] <= d.value <= LEAN_MM[1]]
        size = {"length_mm": max(horiz) if horiz else None, "width_mm": max(vert) if vert else None,
                     "height_mm": max(heights) if heights else None,
                     "lean_thickness_mm": max(Counter(leans).items(), key=lambda kv: (kv[1], kv[0]))[0] if leans else None,
                     "nguon": f"{v.title} tại ({v.x:g}, {v.y:g}), {len(sections.get(k, []))} mặt cắt"}
        previous = out.get(k[1])
        if previous and (previous.get("conflict") or any(size[d] != previous[d] for d in size if d != "nguon")):
            def describe(s):
                return " × ".join("?" if s[d] is None else f"{s[d]:g}" for d in ("length_mm", "width_mm", "height_mm"))
            earlier = previous["nguon"] if previous.get("conflict") else f"{previous['nguon']} ({describe(previous)} mm)"
            size = {d: None for d in size if d != "nguon"} | {
                "conflict": True, "nguon": f"{earlier} / {size['nguon']} ({describe(size)} mm)"}
        out[k[1]] = size
    return out

def read_structural(drawing, scale: float | None = None, dims: list[Dim] | None = None) -> StructuralTakeoff:
    """scale: hệ số đổi số đo trên mặt bằng kết cấu ra mm thật; None: tự suy từ đường kích thước của từng mặt bằng.
    dims: đường kích thước (mặc định đọc từ bản vẽ gốc)."""
    views = find_views(drawing, bbox_for=PLAN_TYPES + ("BV05",))
    out = StructuralTakeoff(views=views, sheets=sheet_inventory(views))
    from .structural_column_instances import read_column_instances
    out.structural_columns = read_column_instances(drawing, views)
    dims = read_dimensions(drawing) if dims is None else dims
    details = detail_sections(drawing, views, dims)
    beam_sizes = beam_detail_sizes(details)
    if beam_sizes:
        out.warnings.append("Tiết diện dầm đọc từ mặt cắt dầm (rộng × cao gồm sàn): "
                            + ", ".join(f"{k.upper()} {b:.0f}×{h:.0f}" for k, (b, h) in sorted(beam_sizes.items()))
                            + "; kiểm tra trên trang kiểm tra.")
    sections = {**beam_sizes, **written_sections(drawing)}      # tiết diện ghi chữ thắng
    spots: dict[str, list] = {}             # móng -> [(mặt bằng, x, y, tỷ lệ)] của từng ký hiệu
    plan_lines: dict[str, list] = {}
    for v in views:
        if v.ma not in PLAN_TYPES or v.bbox is None:
            continue
        s = scale
        if s is None:
            s = view_scale(inside(dims, _grown(v.bbox)))
            if v.ma in ("BV04", "BV08") and (picked := _scale_by_beam_width(drawing, v, inside(dims, _grown(v.bbox)), sections)):

                                        "cho bề rộng dầm đo được khớp bề rộng ghi trên mặt cắt dầm, kiểm tra.")
                s = picked
            if s is None:
                s = 1.0
                if v.ma in ("BV04", "BV08"):
                    out.warnings.append(f"{v.title}: không có đường kích thước để suy tỷ lệ, đang coi số đo là mm thật.")
            elif abs(s - 1) > 0.01 and v.ma in ("BV04", "BV08"):
                out.warnings.append(f"{v.title}: tỷ lệ hình vẽ {s:.3g} (số hiển thị / số đo của đường kích thước), "
                                    "chiều dài, rộng dầm đã đổi theo tỷ lệ này.")
        inside_labels = _labels(drawing, v.bbox)
        cols = Counter(t.text.strip().upper() for t, e in inside_labels if e[0] == "COLUMN")
        if cols:
            out.columns[v.title] = dict(sorted(cols.items()))
        if v.ma == "BV04":
            for t, e in inside_labels:
                if e[0] == "FOUNDATION":
                    spots.setdefault(t.text.strip().upper(), []).append((v.title, t.x, t.y, s))
            plan_lines[v.title] = _lines(drawing, _grown(v.bbox))
        if v.ma in ("BV04", "BV08"):
            beams, warn = measure_beams(drawing, v, sections, s)
            if beams:
                out.beams[v.title] = beams
            out.warnings += warn
    out.column_sections = {k: v for k, v in sections.items() if (e := element_of(k)) and e[0] == "COLUMN"}
    out.column_schedule, warn = read_column_schedule(drawing, dims)
    out.warnings += warn
    sizes = footing_detail_sizes(views, details, dims)
    strip_sections = {_title_key(v.title) for v in views if v.ma == "BV05" and "mong bang" in normalize(v.title)}
    strip_notes: dict[str, set[tuple[float, int | None]]] = {}
    for t in drawing.texts:
        note = re.fullmatch(r"(bm[-–]?\d+[a-z]?)\s*:\s*l\s*=\s*(\d+(?:[.,]\d+)?)\s*mm(?:\s*;\s*sl\s*=\s*(\d+))?",
                            t.text.strip(), re.IGNORECASE)
        if note and (mark := element_of(note[1])) and ("FOUNDATION", mark[2]) in strip_sections:
            length = float(note[2].replace(",", "."))
            if length > 0:
                strip_notes.setdefault(mark[2], set()).add((length, int(note[3]) if note[3] else None))
    for n, pts in sorted(spots.items()):
        mark = element_of(n)[2]
        size = sizes.get(mark, {})
        is_strip = mark.startswith("bm")
        if is_strip:
            lengths = {length for length, _ in strip_notes.get(mark, set())}
            if len(lengths) == 1:
                length = next(iter(lengths))
                size = {"gross_length_mm": length, "nguon": f"ghi chú {n}: L={length:g} mm và mặt cắt móng băng cùng ký hiệu"}

        for p in pts:
            if not any(q[0] == p[0] and math.hypot(q[1] - p[1], q[2] - p[2]) * p[3] < SAME_SPOT_MM for q in uniq):
                uniq.append(p)
        c = len(uniq)
        if len(pts) > c:
            out.warnings.append(f"Móng {n}: {len(pts) - c} ký hiệu trùng vị trí, đếm một lần.")
        centers = []
        if len(plans) == 1 and size.get("length_mm") and size.get("width_mm"):
            for title, x, y, sc in uniq:
                ctr = _footing_center(plan_lines[title], x, y, size["length_mm"] / sc, size["width_mm"] / sc)
                centers.append(None if ctr is None else (round(ctr[0] * sc, 1), round(ctr[1] * sc, 1)))
        found_all = bool(centers) and None not in centers
        if size.get("conflict"):
            out.warnings.append(f"Móng {n}: các chi tiết cùng ký hiệu có kích thước khác nhau; cần xác định chi tiết cho từng vị trí. {size['nguon']}")
        f = FoundationItem(name=n, count=c, kind="bang" if is_strip else None, **{k: v for k, v in size.items() if k.endswith("_mm")},
                           positions=centers if found_all else [], plan=next(iter(plans)) if found_all else None,
                           source=f"{n} trên {', '.join(sorted(plans))}" + (f"; kích thước: {size['nguon']}" if size else ""))
        if centers and not found_all:
            out.warnings.append(f"Móng {n}: không tìm thấy khung móng {size['length_mm']:.0f}×{size['width_mm']:.0f} quanh "
                                f"{centers.count(None)}/{c} ký hiệu: chưa có vị trí để kiểm tra hố đào chồng lấn, giằng qua móng.")
        out.foundations.append(f)
        dims_txt = " × ".join(f"{v:.0f}" if v is not None else "?" for v in (f.length_mm, f.width_mm, f.height_mm))
        out.warnings.append(f"Móng {n}: {c} khối, {dims_txt} mm"
                            + (f", lót {f.lean_thickness_mm:.0f} mm" if f.lean_thickness_mm else "")
                            + (" (móng băng: chờ kiểm tra tiết diện và phần giao)." if is_strip else
                               " (chờ đối chiếu các chi tiết cùng ký hiệu)." if size.get("conflict") else
                               f" (đọc từ chi tiết {size['nguon']}; móng coi là khối hộp, móng có bậc hoặc vát cần sửa)."
                               if size else " (không có chi tiết móng: nhập kích thước trong thông số công trình)."))
    located_marks = {element_of(n)[2] for n in spots}
    for mark, notes in sorted(strip_notes.items()):
        if mark in located_marks:
            continue
        if len(notes) != 1:
            out.warnings.append(f"{mark.upper()}: các ghi chú L/SL khác nhau, cần đối chiếu từng vị trí trên mặt bằng.")
            continue
        length, count = next(iter(notes))
        if not count:
            out.warnings.append(f"{mark.upper()}: đọc được L={length:g} mm, thiếu số lượng và vị trí trên mặt bằng.")
            continue
        name = f"BM-{mark[2:].upper()}"
        source = f"Chi tiết móng băng {name}, L={length:g} mm, SL={count}; chưa nối vị trí trên mặt bằng"
        out.foundations.append(FoundationItem(name=name, kind="bang", count=count, gross_length_mm=length, source=source))
        out.warnings.append(f"{source}. Chưa tính thể tích: thiếu tiết diện và phạm vi giao nhau.")
    return out

def find_views(drawing: Drawing, bbox_for: tuple[str, ...] | None = None) -> list[View]:
    """Hình vẽ có tiêu đề nhận được loại. bbox_for: chỉ tìm khung cho các loại này (tìm khung tốn thời gian); None: mọi loại."""
    titled = [(t, st) for t in drawing.texts if (st := sheet_type_of(t.text))]
    if not titled:
        return []
    big = title_height_reference([t for t, _ in titled])
    titled = [(t, st) for t, st in titled if t.height >= TITLE_REL_HEIGHT * big]
    headings = [t for t, _ in titled]
    views, seen = [], Counter()
    for t, st in sorted(titled, key=lambda p: (-p[0].y, p[0].x)):
        title = re.sub(r"\s+", " ", t.text).strip()
        seen[title] += 1      # QU-05: tiêu đề là khóa (dầm theo mặt bằng), trùng thì đánh số thêm
        views.append(View(title if seen[title] == 1 else f"{title} ({seen[title]})", st.ma, st.ten, st.nhom, t.x, t.y,
                          _region(drawing, t, headings) if bbox_for is None or st.ma in bbox_for else None))
    return views

def title_height_reference(titles) -> float:
    # Tên cấu kiện đơn lẻ (ví dụ chữ “lanh tô” phóng lớn) không đại diện cỡ tiêu đề mặt bằng.
    headings = [t.height for t in titles if normalize(t.text).startswith(HEADING_WORDS) and t.height > 0]
    return max(headings, default=0) or max((t.height for t in titles), default=0)

def build_registry(sources: list[tuple[str, Drawing]], symbol_overrides: dict[str, str] | None = None,
                   dims: dict[str, list] | None = None) -> Registry:
    """sources: [(tên file, bản vẽ đã đọc)], kiến trúc và kết cấu chung một hồ sơ. dims: đường kích thước theo file

        all_views += views

        if revs:
            revisions[file] = revs
        for name in d.layers:
            sig = match_layer(normalize(name))
            role = d.layer_classes.get(name, "UNKNOWN")
            layers.append(LayerMap(file=file, layer=name, role=getattr(role, "value", role),
                                   rule=sig.ma if sig else None))

                unmatched.append(Unmatched(kind="layer", text=name))
        for p in find_plans(d):          # mặt bằng kiến trúc: cao độ tầng
            for key in floors_of(p.title or p.name):
                f = floors.setdefault(key, Floor(key=key, names=[]))
                f.names.append(p.name)
                f.elevation_m = f.elevation_m if f.elevation_m is not None else p.level_m
        # tiêu đề nhận được loại -> ViewInfo; chữ cỡ tiêu đề không nhận được loại -> unmatched
        title_texts = {(round(v.x), round(v.y)) for v in views}
        big = title_height_reference([t for t in d.texts if sheet_type_of(t.text)])
        for t in d.texts:
            n = normalize(t.text).strip()

                    and n.startswith(TITLE_WORDS) and not sheet_type_of(t.text):

        by_pos = {(round(t.x), round(t.y)): t for t in d.texts}
        for v in views:
            t = by_pos.get((round(v.x), round(v.y)))

                                                                   text=v.title)
            if v.bbox:
                ev.bbox = tuple(round(c, 1) for c in v.bbox)
            fl = floors_of(v.title)
            scale = view_scale(inside(fdims, v.bbox)) if v.bbox else None
            views_out.append(ViewInfo(file=file, title=v.title, sheet_type=v.ma, sheet_name=v.ten, floors=fl,
                                      scale=scale, origin=(v.bbox[0], v.bbox[1]) if v.bbox else None, evidence=ev))
            for key in fl:
                floors.setdefault(key, Floor(key=key, names=[])).views.append(v.title)
            if v.ma in DETAIL_TYPES:
                keyed = False
                for word in re.split(r"[\s(),:]+", v.title):

                    if e and e[0] in REQUIRED:

                    typical.setdefault(TYPICAL_ROLE[v.ma], []).append(ev)

        for cid, sv in _written(file, d, overrides, handles):

            comps.setdefault(cid, _new(cid, role, role.lower(), key.upper())).size_sources.append(sv)

        details = st.detail_sections(d, views, fdims)
        for key, (b, h) in st.beam_detail_sizes(details).items():

        for key, size in st.footing_detail_sizes(views, details, fdims).items():

            c = comps.setdefault(cid, _new(cid, "FOUNDATION", "móng", key.upper()))
            if size.get("conflict"):
                issues.append(Issue(kind="xung_dot_kich_thuoc", component=cid,
                                    message=f"Các chi tiết móng cùng ký hiệu khác kích thước: {size['nguon']}"))
            vals = {k: v for k, v in size.items() if k in REQUIRED["FOUNDATION"] + ("lean_thickness_mm",) and v is not None}
            if vals:
                ev = c.details[0] if c.details else EvidenceRef(file=file, bbox=(0, 0, 0, 0), rule="CHI_TIET")
                c.size_sources.append(SizeValue(values=vals, source="chi_tiet", evidence=ev))
    for c in comps.values():
        if not c.details:
            c.details = list(typical.get(c.role, []))
    _floor_issues(floors, views_out, issues)
    if len({r for rs in revisions.values() for r in rs}) > 1:
        issues.append(Issue(kind="lech_phien_ban", message="Phiên bản ghi trên bản vẽ không thống nhất: "
                            + "; ".join(f"{f}: {', '.join(r)}" for f, r in revisions.items())))
    reg = Registry(files=[f for f, _ in sources], sheets=sheet_inventory(all_views), views=views_out,
                   floors=sorted(floors.values(), key=lambda f: f.key), layers=layers,
                   components=sorted(comps.values(), key=lambda c: c.id), issues=issues, unmatched=unmatched,
                   revisions=revisions)
    for c in reg.components:
        _resolve(c, reg)
    return score(reg)
